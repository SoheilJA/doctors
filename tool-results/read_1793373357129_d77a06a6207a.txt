    97→  "bg-primary/8",
    98→  "bg-primary/15",
    99→  "bg-primary/25",
   100→  "bg-primary/40",
   101→  "bg-primary/55",

   106→const HEAT_HOURS = Array.from({ length: 14 }, (_, i) => i + 8); // 8..21
   107→
   108→export default function DoctorDashboardPage() {
   109→  const loading = useDemoLoading();
   110→  const router = useRouter();
   111→  const { doctor, appointments, consultations, reviews, patientMap } = useDoctorPanel();
   112→  const setAppointmentStatus = useAppStore((s) => s.setAppointmentStatus);
   113→  const [cancelTarget, setCancelTarget] = useState<Appointment | null>(null);
   114→
   115→  const today = todayISO();
   116→
   117→  /* ------------------------------ stats ------------------------------ */
   118→
   119→  const stats = useMemo(() => {
   120→    const todays = appointments.filter((a) => a.date === today).sort(byTime);
   121→    const weekStart = startOfWeekISO(today);
   122→    const weekEnd = endOfWeekISO(today);
   123→    const thisWeek = appointments.filter(
   124→      (a) => a.date >= weekStart && a.date <= weekEnd && a.status !== "cancelled" && a.status !== "no-show"
   125→    );
   126→    const monthKey = jalaliMonthKey(today);
   127→    const monthCompleted = appointments.filter(
   128→      (a) => a.status === "completed" && a.paymentStatus === "paid" && jalaliMonthKey(a.date) === monthKey
   129→    );
   130→    const monthNet = monthCompleted.reduce((s, a) => s + netFee(a), 0);
   131→    const waiting = consultations.filter((c) => c.status === "waiting");
   132→    const overdue = waiting.filter((c) => new Date(c.deadlineISO).getTime() < Date.now());
   133→
   134→    // patients whose first-ever visit with this doctor happened this month
   135→    const firstVisit = new Map<string, string>();
   136→    for (const a of [...appointments].sort(byDateTimeAsc)) {
   137→      if (a.status === "cancelled") continue;
   138→      if (!firstVisit.has(a.patientId)) firstVisit.set(a.patientId, a.date);
   139→    }
   140→    const newPatients = [...firstVisit.values()].filter((d) => jalaliMonthKey(d) === monthKey).length;
   141→
   142→    return {
   143→      todays,
   144→      confirmedToday: todays.filter((a) => a.status === "confirmed").length,
   145→      thisWeek: thisWeek.length,
   146→      monthNet,
   147→      monthVisitCount: monthCompleted.length,
   148→      waiting: waiting.length,
   149→      overdue: overdue.length,
   150→      newPatients,
   151→    };
   152→  }, [appointments, consultations, today]);
   153→
   154→  /* ------------------------------ charts ------------------------------ */
   155→
   156→  const visits30 = useMemo(() => {
   157→    const byDay = new Map<string, number>();
   158→    for (const a of appointments) {
   159→      if (a.status !== "completed") continue;
   160→      byDay.set(a.date, (byDay.get(a.date) ?? 0) + 1);
   161→    }

   163→      label: faDateShort(d),
   164→      visits: byDay.get(d) ?? 0,
   165→    }));
   166→  }, [appointments, today]);
   167→
   168→  const monthKeys = useMemo(
   169→    () => [5, 4, 3, 2, 1, 0].map((i) => jalaliMonthKey(subMonthsISO(today, i))),
   170→    [today]
   171→  );
   172→
   173→  const income6 = useMemo(
   174→    () =>
   175→      monthKeys.map((key) => {
   176→        const rows = appointments.filter(
   177→          (a) => a.status === "completed" && a.paymentStatus === "paid" && jalaliMonthKey(a.date) === key
   178→        );
   179→        return {
   180→          label: faMonthLabel(key),
   181→          net: rows.reduce((s, a) => s + netFee(a), 0),
   182→          gross: rows.reduce((s, a) => s + a.fee, 0),
   183→        };
   184→      }),
   185→    [appointments, monthKeys]
   186→  );
   187→
   188→  const visitTypeData = useMemo(() => {
   189→    const keys: VisitType[] = ["in-person", "phone", "text", "video"];
   190→    return keys
   191→      .map((k) => ({
   192→        key: k === "in-person" ? "inPerson" : k,
   193→        label: VISIT_TYPE_LABELS[k],
   194→        value: appointments.filter((a) => a.visitType === k && a.status !== "cancelled").length,
   195→      }))
   196→      .filter((d) => d.value > 0);
   197→  }, [appointments]);
   198→

   200→    const counts = new Map<string, number>();
   201→    let max = 1;
   202→    for (const a of appointments) {
   203→      if (a.status === "cancelled" || a.status === "no-show") continue;
   204→      const key = `${weekdayIndex(a.date)}|${Number(a.time.split(":")[0])}`;
   205→      const v = (counts.get(key) ?? 0) + 1;
   206→      counts.set(key, v);
   207→      max = Math.max(max, v);
   208→    }
   209→    return { counts, max };
   210→  }, [appointments]);
   211→
   212→  const ratingTrend = useMemo(
   213→    () =>
   214→      monthKeys.map((key) => {
   215→        const rows = reviews.filter((r) => jalaliMonthKey(r.createdAtISO) === key);
   216→        return {
   217→          label: faMonthLabel(key),
   218→          avg: rows.length ? Math.round((rows.reduce((s, r) => s + r.rating, 0) / rows.length) * 10) / 10 : null,
   219→        };
   220→      }),
   221→    [monthKeys, reviews]
   222→  );
   223→
   224→  /* --------------------------- lists --------------------------- */
   225→
   226→  const upcoming = useMemo(
   227→    () =>
   228→      appointments
   229→        .filter((a) => (a.status === "confirmed" || a.status === "pending") && a.date >= today)
   230→        .sort(byDateTimeAsc)
   231→        .slice(0, 5),
   232→    [appointments, today]
   233→  );
   234→
   235→  const latestReviews = useMemo(
   236→    () => [...reviews].sort((a, b) => b.createdAtISO.localeCompare(a.createdAtISO)).slice(0, 3),
   237→    [reviews]
   238→  );
   239→

   241→    return <PageHeader title="داشبورد پزشک" description="در حال بارگذاری…" loading={loading} />;
   242→  }

   246→    toast.success(status === "confirmed" ? "نوبت تأیید شد" : "ویزیت انجام‌شده ثبت شد", {
   247→      description: `نوبت ${toFa(a.code)}`,
   248→    });
   249→  };
   250→
   251→  const doCancel = () => {
   252→    if (!cancelTarget) return;
   253→    const wasPaid = cancelTarget.paymentStatus === "paid";
   254→    setAppointmentStatus(cancelTarget.id, "cancelled", doctor.name);
   255→    toast("نوبت لغو شد", { description: wasPaid ? "مبلغ پرداختی به بیمار بازگشت داده شد." : undefined });

   257→  };
   258→
   259→  const visitsConfig = { visits: { label: "ویزیت", color: "var(--chart-1)" } } as const;
   260→  const incomeConfig = { net: { label: "درآمد خالص", color: "var(--chart-2)" } } as const;
   261→  const donutConfig = {
   262→    inPerson: { label: "حضوری", color: "var(--chart-1)" },
   263→    phone: { label: "تلفنی", color: "var(--chart-2)" },
   264→    text: { label: "متنی", color: "var(--chart-3)" },
   265→    video: { label: "ویدیویی", color: "var(--chart-5)" },
   266→  } as const;

   270→    <div className="space-y-6">
   271→      <PageHeader
   272→        title={`سلام ${doctor.name} 👋`}
   273→        description={`امروز ${faDate(today)} — ${toFa(stats.todays.length)} نوبت در برنامه دارید.`}
   274→        loading={loading}
   275→        actions={
   276→          <Button asChild variant="outline" size="sm">
   277→            <Link href="/doctor/calendar">
   278→              <CalendarDays className="size-4" aria-hidden />
   279→              مشاهده تقویم
   280→            </Link>
   281→          </Button>
   282→        }
   283→      />
   284→
   285→      {/* KPI cards */}
   286→      <section className="grid grid-cols-2 gap-3 md:grid-cols-3 xl:grid-cols-6" aria-label="شاخص‌های کلیدی">
   287→        <StatCard
   288→          title="نوبت‌های امروز"
   289→          value={faNum(stats.todays.length)}
   290→          hint={`${toFa(stats.confirmedToday)} نوبت تأییدشده`}
   291→          icon={CalendarCheck}
   292→          tone="primary"
   293→          loading={loading}
   294→          onClick={() => router.push("/doctor/calendar")}
   295→        />
   296→        <StatCard
   297→          title="ویزیت‌های این هفته"
   298→          value={faNum(stats.thisWeek)}
   299→          hint="شنبه تا جمعه جاری"
   300→          icon={ClipboardCheck}
   301→          tone="sky"
   302→          loading={loading}
   303→          onClick={() => router.push("/doctor/appointments")}
   304→        />
   305→        <StatCard
   306→          title="درآمد ماه جاری"
   307→          value={tomanCompact(stats.monthNet)}
   308→          hint={`از ${toFa(stats.monthVisitCount)} ویزیت (پس از کسر کمیسیون)`}
   309→          icon={Wallet}
   310→          tone="emerald"
   311→          loading={loading}
   312→          onClick={() => router.push("/doctor/finance")}

   314→        <StatCard
   315→          title="میانگین امتیاز"
   316→          value={`${toFa(doctor.rating.toFixed(1))} از ۵`}

   318→          icon={Star}
   319→          tone="amber"
   320→          loading={loading}
   321→          onClick={() => router.push("/doctor/reviews")}
   322→        />
   323→        <StatCard
   324→          title="مشاوره‌های بی‌پاسخ"
   325→          value={faNum(stats.waiting)}
   326→          hint="در انتظار پاسخ شما"
   327→          trend={stats.overdue > 0 ? { value: `${toFa(stats.overdue)} مورد مهلت گذشته`, positive: false } : undefined}
   328→          icon={MessageSquareText}
   329→          tone={stats.overdue > 0 ? "rose" : "violet"}
   330→          loading={loading}
   331→          onClick={() => router.push("/doctor/consultations")}
   332→        />
   333→        <StatCard
   334→          title="بیماران جدید این ماه"
   335→          value={faNum(stats.newPatients)}
   336→          hint="اولین ویزیتشان با شما"
   337→          icon={UsersRound}
   338→          tone="violet"
   339→          loading={loading}
   340→          onClick={() => router.push("/doctor/patients")}
   341→        />
   342→      </section>
   343→
   344→      {/* charts row 1 */}
   345→      <section className="grid gap-4 lg:grid-cols-3" aria-label="نمودارها">
   346→        <ChartCard
   347→          title="ویزیت‌های ۳۰ روز گذشته"
   348→          description="تعداد ویزیت‌های انجام‌شده در هر روز"
   349→          config={visitsConfig}
   350→          loading={loading}
   351→          className="lg:col-span-2"
   352→        >
   353→          <AreaChart data={visits30} margin={{ top: 8, left: 0, right: 8 }}>
   354→            <defs>
   355→              <linearGradient id="visitsFill" x1="0" y1="0" x2="0" y2="1">
   356→                <stop offset="5%" stopColor="var(--color-visits)" stopOpacity={0.35} />
   357→                <stop offset="95%" stopColor="var(--color-visits)" stopOpacity={0.03} />
   358→              </linearGradient>
   359→            </defs>
   360→            <CartesianGrid vertical={false} strokeDasharray="3 3" />
   361→            <XAxis dataKey="label" tickLine={false} axisLine={false} tickMargin={8} interval={4} reversed />
   362→            <YAxis tickLine={false} axisLine={false} width={28} allowDecimals={false} tickFormatter={(v) => faNum(Number(v))} />
   363→            <ChartTooltip
   364→              content={
   365→                <ChartTooltipContent
   366→                  labelFormatter={(l) => String(l)}
   367→                  formatter={(v) => [`${faNum(Number(v))} ویزیت`, "انجام‌شده"]}
   368→                />
   369→              }
   370→            />
   371→            <Area dataKey="visits" type="monotone" stroke="var(--color-visits)" strokeWidth={2.2} fill="url(#visitsFill)" />
   372→          </AreaChart>
   373→        </ChartCard>
   374→
   375→        <ChartCard
   376→          title="پرتقاضاترین ساعت‌ها"
   377→          description="تعداد نوبت‌ها به تفکیک روز هفته و ساعت"
   378→          config={{}}
   379→          loading={loading}
   380→          height="h-64"
   381→        >
   382→          <div className="flex h-full flex-col gap-1 overflow-x-auto" dir="rtl">
   383→            <div className="grid grid-cols-8 gap-1 text-center text-[10px] font-medium text-muted-foreground">
   384→              <span />
   385→              {JALALI_WEEKDAYS_SHORT.map((w) => (
   386→                <span key={w}>{w}</span>
   387→              ))}
   388→            </div>
   389→            <div className="grid flex-1 grid-flow-row grid-cols-8 content-stretch gap-1">
   390→              {HEAT_HOURS.map((h) => (
   391→                <div key={h} className="col-span-8 grid grid-cols-8 items-stretch gap-1">
   392→                  <span className="self-center text-[10px] tabular-nums text-muted-foreground">{faTime(`${String(h).padStart(2, "0")}:00`)}</span>
   393→                  {[0, 1, 2, 3, 4, 5, 6].map((w) => {

   395→                    const intensity = count === 0 ? 0 : Math.max(1, Math.ceil((count / heat.max) * HEAT_LADDER.length));
   396→                    return (
   397→                      <div
   398→                        key={w}
   399→                        title={`${JALALI_WEEKDAYS_SHORT[w]} ساعت ${faTime(`${String(h).padStart(2, "0")}:00`)} — ${toFa(count)} نوبت`}
   400→                        className={`min-h-4 rounded-[4px] ${count === 0 ? "bg-muted" : HEAT_LADDER[intensity - 1]}`}
   401→                      />
   402→                    );
   403→                  })}
   404→                </div>
   405→              ))}
   406→            </div>
   407→          </div>
   408→        </ChartCard>
   409→      </section>
   410→
   411→      {/* charts row 2 */}
   412→      <section className="grid gap-4 lg:grid-cols-3" aria-label="نمودار درآمد و نظرات">
   413→        <ChartCard title="درآمد ۶ ماه اخیر" description="سهم خالص شما پس از کسر کمیسیون" config={incomeConfig} loading={loading}>
   414→          <BarChart data={income6} margin={{ top: 8, left: 0, right: 8 }}>
   415→            <CartesianGrid vertical={false} strokeDasharray="3 3" />
   416→            <XAxis dataKey="label" tickLine={false} axisLine={false} tickMargin={8} reversed />
   417→            <YAxis
   418→              tickLine={false}
   419→              axisLine={false}
   420→              width={52}
   421→              tickFormatter={(v) => tomanCompact(Number(v)).replace(" تومان", "")}
   422→            />
   423→            <ChartTooltip
   424→              content={
   425→                <ChartTooltipContent
   426→                  labelFormatter={(l) => String(l)}
   427→                  formatter={(v, _n, item) => [
   428→                    toman(Number(v)),
   429→                    `ناخالص: ${toman(Number(item?.payload?.gross ?? 0))} • خالص`,
   430→                  ]}
   431→                />
   432→              }
   433→            />
   434→            <Bar dataKey="net" fill="var(--color-net)" radius={6} />
   435→          </BarChart>
   436→        </ChartCard>
   437→
   438→        <ChartCard
   439→          title="نوع ویزیت‌ها"
   440→          description="سهم هر نوع ویزیت از نوبت‌های ثبت‌شده"
   441→          config={donutConfig}
   442→          loading={loading}
   443→          legend={
   444→            <div className="flex flex-wrap items-center justify-center gap-x-4 gap-y-1.5 pt-1 text-xs text-muted-foreground">

   455→            </div>

   460→            <Pie data={visitTypeData} dataKey="value" nameKey="key" innerRadius={52} outerRadius={82} paddingAngle={3} strokeWidth={0}>
   461→              {visitTypeData.map((d) => (
   462→                <Cell key={d.key} fill={`var(--color-${d.key})`} />
   463→              ))}
   464→            </Pie>
   465→          </PieChart>
   466→        </ChartCard>
   467→
   468→        <ChartCard title="روند امتیاز" description="میانگین ماهانه امتیاز نظرات" config={ratingConfig} loading={loading}>
   469→          <LineChart data={ratingTrend} margin={{ top: 8, left: 0, right: 8 }}>
   470→            <CartesianGrid vertical={false} strokeDasharray="3 3" />
   471→            <XAxis dataKey="label" tickLine={false} axisLine={false} tickMargin={8} reversed interval={1} />
   472→            <YAxis domain={[0, 5]} tickLine={false} axisLine={false} width={24} tickFormatter={(v) => toFa(Number(v))} />
   473→            <ChartTooltip

   475→                <ChartTooltipContent

   481→            <Line dataKey="avg" type="monotone" stroke="var(--color-avg)" strokeWidth={2.4} dot={{ r: 3 }} connectNulls />

   486→      {/* today program + upcoming */}

   488→        <Card className="lg:col-span-2">

   491→              <CardTitle className="text-base">برنامه امروز</CardTitle>

   499→            </Button>

   501→          <CardContent>
