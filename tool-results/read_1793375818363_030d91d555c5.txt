   203→/* ------------------------------ helpers ------------------------------ */
   204→
   205→const genId = (prefix: string) => `${prefix}-${Math.random().toString(36).slice(2, 8)}${Date.now() % 1000}`;
   206→let codeSeq = Math.floor(Math.random() * 4000) + 90000;
   207→const genCode = (prefix: string) => `${prefix}-${++codeSeq}`;

   209→/* ------------------------------ store ------------------------------ */
   210→
   211→const seed = buildSeed();
   212→
   213→export const useAppStore = create<AppStore>()(
   214→  persist(
   215→    (set, get) => ({
   216→      ...seed,
   217→      _hasHydrated: false,
   218→      setHasHydrated: (v) => set({ _hasHydrated: v }),
   219→
   220→      /* ------------------------------ meta ------------------------------ */
   221→      addAuditLog: (entry) =>
   222→        set((s) => ({
   223→          auditLog: [{ id: genId("aud"), atISO: nowISODateTime(), ...entry }, ...s.auditLog],
   224→        })),
   225→
   226→      resetDemoData: () => {
   227→        const fresh = buildSeed();
   228→        set({ ...fresh });
   229→      },
   230→
   231→      /* -------------------------- notifications -------------------------- */
   232→      pushNotification: (n) =>
   233→        set((s) => ({
   234→          notifications: [
   235→            { id: genId("ntf"), read: false, createdAtISO: nowISODateTime(), ...n },
   236→            ...s.notifications,
   237→          ],
   238→        })),
   239→      markNotificationRead: (id) =>
   240→        set((s) => ({
   241→          notifications: s.notifications.map((n) => (n.id === id ? { ...n, read: true } : n)),
   242→        })),
   243→      markAllNotificationsRead: (role, userId) =>
   244→        set((s) => ({
   245→          notifications: s.notifications.map((n) =>
   246→            n.role === role && (n.userId ? n.userId === userId || n.userId === undefined : true) ? { ...n, read: true } : n
   247→          ),
   248→        })),
   249→
   250→      /* --------------------------- appointments --------------------------- */
   251→      bookAppointment: (input) => {
   252→        const doctor = get().doctors.find((d) => d.id === input.doctorId)!;
   253→        const feeMap: Record<VisitType, number> = {
   254→          "in-person": doctor.visitFee,
   255→          phone: doctor.phoneFee,
   256→          text: doctor.textFee,
   257→          video: doctor.videoFee,
   258→        };
   259→        const fee = feeMap[input.visitType];
   260→        const appointment: Appointment = {
   261→          id: genId("apt"),
   262→          code: genCode("NO"),
   263→          doctorId: input.doctorId,
   264→          patientId: input.patientId,
   265→          forFamilyMember: input.forFamilyMember,
   266→          date: input.date,
   267→          time: input.time,
   268→          duration: input.visitType === "in-person" ? 15 : 20,
   269→          visitType: input.visitType,
   270→          status: input.payNow ? "confirmed" : "pending",
   271→          paymentStatus: input.payNow ? "paid" : "unpaid",
   272→          fee,
   273→          commission: Math.round(fee * (doctor.commissionRate / 100)),
   274→          reason: input.reason ?? "",
   275→          notes: "",
   276→          createdAtISO: nowISODateTime(),
   277→        };
   278→        const transaction: Transaction | null = input.payNow
   279→          ? {
   280→              id: genId("trx"),
   281→              code: genCode("TR"),
   282→              kind: input.visitType === "in-person" ? "appointment" : "consultation",
   283→              amount: fee,
   284→              direction: "in",
   285→              patientId: input.patientId,
   286→              doctorId: input.doctorId,
   287→              appointmentId: appointment.id,
   288→              status: "success",
   289→              description: `پرداخت نوبت ${appointment.code}`,
   290→              createdAtISO: nowISODateTime(),
   291→            }
   292→          : null;
   293→        set((s) => ({
   294→          appointments: [appointment, ...s.appointments],
   295→          transactions: transaction ? [transaction, ...s.transactions] : s.transactions,
   296→        }));
   297→        get().pushNotification({
   298→          role: "doctor",
   299→          userId: doctor.id,
   300→          title: "نوبت جدید ثبت شد",
   301→          body: `برای ${appointment.date} ساعت ${appointment.time} نوبت جدید ثبت شد.`,
   302→          link: "/doctor/calendar",
   303→        });
   304→        return appointment;
   305→      },
   306→
   307→      updateAppointment: (id, patch) =>
   308→        set((s) => ({ appointments: s.appointments.map((a) => (a.id === id ? { ...a, ...patch } : a)) })),
   309→
   310→      setAppointmentStatus: (id, status, actorName) => {
   311→        const appointment = get().appointments.find((a) => a.id === id);
   312→        set((s) => ({ appointments: s.appointments.map((a) => (a.id === id ? { ...a, status } : a)) }));
   313→        if (!appointment) return;
   314→        // refund side-effect on cancel of a paid appointment
   315→        if (status === "cancelled" && appointment.paymentStatus === "paid") {
   316→          const refund: Transaction = {
   317→            id: genId("trx"),
   318→            code: genCode("TR"),
   319→            kind: "refund",
   320→            amount: appointment.fee,
   321→            direction: "out",
   322→            patientId: appointment.patientId,
   323→            doctorId: appointment.doctorId,
   324→            appointmentId: appointment.id,
   325→            status: "success",
   326→            description: `بازگشت وجه نوبت لغوشده ${appointment.code}`,
   327→            createdAtISO: nowISODateTime(),
   328→          };
   329→          set((s) => ({
   330→            transactions: [refund, ...s.transactions],
   331→            appointments: s.appointments.map((a) => (a.id === id ? { ...a, paymentStatus: "refunded" } : a)),
   332→          }));
   333→        }
   334→        if (status === "completed") {
   335→          get().pushNotification({
   336→            role: "patient",
   337→            userId: appointment.patientId,
   338→            title: "ویزیت شما انجام شد",
   339→            body: `نوبت ${appointment.code} با موفقیت انجام شد. می‌توانید نظر خود را ثبت کنید.`,
   340→            link: "/patient/appointments",
   341→          });
   342→        }
   343→      },
   344→
   345→      rescheduleAppointment: (id, date, time) => {
   346→        set((s) => ({ appointments: s.appointments.map((a) => (a.id === id ? { ...a, date, time, status: "pending" } : a)) }));
   347→        const appointment = get().appointments.find((a) => a.id === id);
   348→        if (appointment) {
   349→          get().pushNotification({
   350→            role: "doctor",
   351→            userId: appointment.doctorId,
   352→            title: "تغییر زمان نوبت",
   353→            body: `نوبت ${appointment.code} به ${date} ساعت ${time} تغییر یافت.`,
   354→            link: "/doctor/calendar",
   355→          });
   356→        }
   357→      },
   358→
   359→      /* --------------------------- consultations --------------------------- */
   360→      createConsultation: (input) => {
   361→        const doctor = get().doctors.find((d) => d.id === input.doctorId)!;
   362→        const fee = input.type === "text" ? doctor.textFee : input.type === "phone" ? doctor.phoneFee : doctor.videoFee;
   363→        const createdAtISO = nowISODateTime();
   364→        const consultation = {
   365→          id: genId("con"),
   366→          code: genCode("CO"),
   367→          doctorId: input.doctorId,
   368→          patientId: input.patientId,
   369→          type: input.type,
   370→          status: "waiting" as const,
   371→          subject: input.subject,
   372→          question: input.question,
   373→          messages: [
   374→            {
   375→              id: genId("msg"),
   376→              sender: "patient" as const,
   377→              text: `${input.subject}\n${input.question}`,
   378→              attachments: [],
   379→              sentAtISO: createdAtISO,
   380→            },
   381→          ],
   382→          fee,
   383→          paymentStatus: "paid" as const,
   384→          createdAtISO,
   385→          deadlineISO: addDaysISO(todayISO(), 1) + "T" + createdAtISO.split("T")[1],
   386→        };
   387→        const transaction: Transaction = {
   388→          id: genId("trx"),
   389→          code: genCode("TR"),
   390→          kind: "consultation",
   391→          amount: fee,
   392→          direction: "in",
   393→          patientId: input.patientId,
   394→          doctorId: input.doctorId,
   395→          consultationId: consultation.id,
   396→          status: "success",
   397→          description: `پرداخت مشاوره ${consultation.code}`,
   398→          createdAtISO,
   399→        };
   400→        set((s) => ({ consultations: [consultation, ...s.consultations], transactions: [transaction, ...s.transactions] }));
   401→        get().pushNotification({
   402→          role: "doctor",
   403→          userId: input.doctorId,
   404→          title: "مشاوره جدید",
   405→          body: "یک مشاوره جدید در انتظار پاسخ شماست.",
   406→          link: "/doctor/consultations",
   407→        });
   408→        return { consultation };
   409→      },
   410→
   411→      addConsultationMessage: (consultationId, msg) => {
   412→        const full = {
   413→          id: genId("msg"),
   414→          sender: msg.sender,
   415→          text: msg.text,
   416→          attachments: msg.attachments ?? [],
   417→          sentAtISO: nowISODateTime(),
   418→        };
   419→        set((s) => ({
   420→          consultations: s.consultations.map((c) =>
   421→            c.id === consultationId
   422→              ? {
   423→                  ...c,
   424→                  messages: [...c.messages, full],
   425→                  status: msg.sender === "doctor" && c.status === "waiting" ? "answered" : c.status,
   426→                }
   427→              : c
   428→          ),
   429→        }));
   430→        const consultation = get().consultations.find((c) => c.id === consultationId);
   431→        if (consultation && msg.sender === "doctor") {
   432→          get().pushNotification({
   433→            role: "patient",
   434→            userId: consultation.patientId,
   435→            title: "پاسخ مشاوره",
   436→            body: `پزشک به مشاوره ${consultation.code} پاسخ داد.`,
   437→            link: "/patient/consultations",
   438→          });
   439→        }
   440→      },
   441→
   442→      setConsultationStatus: (id, status) =>
   443→        set((s) => ({ consultations: s.consultations.map((c) => (c.id === id ? { ...c, status } : c)) })),
   444→
   445→      setConsultationPrescription: (id, note) =>
   446→        set((s) => ({ consultations: s.consultations.map((c) => (c.id === id ? { ...c, prescriptionNote: note } : c)) })),
   447→
   448→      /* ------------------------------ reviews ------------------------------ */
   449→      addReview: (input) => {
   450→        const review: Review = {
   451→          id: genId("rev"),
   452→          ...input,
   453→          status: "published",
   454→          createdAtISO: nowISODateTime(),
   455→        };
   456→        set((s) => ({
   457→          reviews: [review, ...s.reviews],
   458→          doctors: s.doctors.map((d) => {
   459→            if (d.id !== input.doctorId) return d;
   460→            const count = d.reviewCount + 1;
   461→            const rating = Math.round(((d.rating * d.reviewCount + input.rating) / count) * 10) / 10;
   462→            return { ...d, reviewCount: count, rating };
   463→          }),
   464→        }));
   465→        get().pushNotification({
   466→          role: "doctor",
   467→          userId: input.doctorId,
   468→          title: "نظر جدید ثبت شد",
   469→          body: "یک نظر جدید برای شما ثبت شد.",
   470→          link: "/doctor/reviews",
   471→        });
   472→      },
   473→
   474→      replyToReview: (id, reply) =>
   475→        set((s) => ({ reviews: s.reviews.map((r) => (r.id === id ? { ...r, reply } : r)) })),
   476→
   477→      setReviewStatus: (id, status, actorName) => {
   478→        set((s) => ({ reviews: s.reviews.map((r) => (r.id === id ? { ...r, status } : r)) }));
   479→        if (actorName && status !== "published") {
   480→          get().addAuditLog({ actorName, actorRole: "admin", action: status === "hidden" ? "مخفی‌سازی نظر" : "بررسی نظر", target: "مدیریت نظرات" });
   481→        }
   482→      },
   483→
   484→      /* ----------------------------- lab orders ----------------------------- */
   485→      createLabOrder: (input) => {
   486→        const totalFee = input.tests.reduce((sum, t) => sum + t.price, 0);
   487→        const order: LabOrder = {
   488→          id: genId("lab"),
   489→          code: genCode("LB"),
   490→          patientId: input.patientId,
   491→          labId: input.labId,
   492→          doctorId: input.doctorId,
   493→          tests: input.tests,
   494→          totalFee,
   495→          status: "registered",
   496→          paymentStatus: "unpaid",
   497→          prescriptionFile: input.prescriptionFile,
   498→          homeSampling: input.homeSampling,
   499→          history: [{ status: "registered", atISO: nowISODateTime() }],
   500→          createdAtISO: nowISODateTime(),
   501→        };
   502→        set((s) => ({ labOrders: [order, ...s.labOrders] }));
   503→        return order;
   504→      },
   505→
   506→      setLabOrderStatus: (id, status) => {
   507→        const order = get().labOrders.find((l) => l.id === id);
   508→        if (!order) return;
   509→        const already = order.history.some((h) => h.status === status);
   510→        set((s) => ({
   511→          labOrders: s.labOrders.map((l) =>
   512→            l.id === id
   513→              ? {
   514→                  ...l,
   515→                  status,
   516→                  paymentStatus: status === "paid" || (l.paymentStatus === "paid") ? "paid" : l.paymentStatus,
   517→                  history: already ? l.history : [...l.history, { status, atISO: nowISODateTime() }],
   518→                  resultNote: status === "received" && !l.resultNote ? "همه مقادیر در محدوده نرمال هستند؛ نتیجه تفصیلی در فایل PDF پیوست است." : l.resultNote,
   519→                }
   520→              : l
   521→          ),
   522→        }));
   523→        if (status === "received") {
   524→          get().pushNotification({
   525→            role: "patient",
   526→            userId: order.patientId,
   527→            title: "جواب آزمایش آماده شد",
   528→            body: `جواب آزمایش «${order.tests[0]?.name}» در بخش آزمایش‌ها قابل مشاهده است.`,
   529→            link: "/patient/labs",
   530→          });
   531→        }
   532→      },
   533→
   534→      payLabOrder: (id) => {
   535→        const order = get().labOrders.find((l) => l.id === id);
   536→        if (!order || order.paymentStatus === "paid") return;
   537→        const transaction: Transaction = {
   538→          id: genId("trx"),
