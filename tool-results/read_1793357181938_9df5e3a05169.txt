    57→/** add N hours to an ISO datetime string, returning an ISO datetime string */
    58→function plusHours(isoDateTime: string, hours: number): string {
    59→  const d = new Date(isoDateTime);
    60→  const shifted = new Date(d.getTime() + hours * 3600_000);
    61→  return `${dateToISO(shifted)}T${String(shifted.getHours()).padStart(2, "0")}:${String(shifted.getMinutes()).padStart(2, "0")}:00`;
