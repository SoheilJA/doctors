   428→/* ------------------------- overrides ------------------------- */
   429→
   430→const overrideSchema = z.object({
   431→  date: z.string().min(1, "تاریخ را انتخاب کنید"),
   432→  note: z.string().trim().optional(),
   433→  ranges: z
   434→    .array(timeRangeSchema)
   435→    .min(1, "حداقل یک بازه ساعات کاری وارد کنید")
   436→    .max(2, "حداکثر دو بازه برای ساعت ویژه مجاز است")
   437→    .superRefine((ranges, ctx) => {
   438→      const sorted = [...ranges].sort((a, b) => timeToMinutes(a.start) - timeToMinutes(b.start));
   439→      for (let i = 0; i < sorted.length; i++) {
   440→        for (let j = i + 1; j < sorted.length; j++) {
   441→          if (timeToMinutes(sorted[j].start) < timeToMinutes(sorted[i].end)) {
   442→            ctx.addIssue({ code: "custom", message: "بازه‌ها نباید همپوشانی داشته باشند؛ لطفاً ساعات را اصلاح کنید" });
   443→            return;
   444→          }
   445→        }
   446→      }
   447→    }),
   448→});

   772→interface AvailabilityEditorProps {
   773→  doctor: Doctor;
   774→  appointments: Appointment[];
   775→  loading: boolean;
   776→  /** triggers a remount so drafts re-initialize from the fresh store */
   777→  onApplied: () => void;
   778→}
   779→
   780→function AvailabilityEditor({ doctor, appointments: allAppointments, loading, onApplied }: AvailabilityEditorProps) {
   781→  const updateDoctorSchedule = useAppStore((s) => s.updateDoctorSchedule);
   782→  const addDoctorOffDay = useAppStore((s) => s.addDoctorOffDay);
   783→  const removeDoctorOffDay = useAppStore((s) => s.removeDoctorOffDay);
   784→  const addDoctorOverride = useAppStore((s) => s.addDoctorOverride);
   785→  const removeDoctorOverride = useAppStore((s) => s.removeDoctorOverride);
   786→
   787→  const [draft, setDraft] = useState<WeeklySchedule>(() => structuredClone(doctor.weeklySchedule));
   788→  const [draftOffDays, setDraftOffDays] = useState<OffDay[]>(() => doctor.offDays.map((o) => ({ ...o })));
   789→  const [draftOverrides, setDraftOverrides] = useState<DateOverride[]>(
   790→    () => doctor.overrides.map((o) => ({ ...o, ranges: o.ranges.map((r) => ({ ...r })) }))
   791→  );
   792→  const [previewDate, setPreviewDate] = useState<string | undefined>(todayISO());
   793→
   794→  const savedOffIds = useMemo(() => new Set(doctor.offDays.map((o) => o.id)), [doctor.offDays]);
   795→  const savedOvrIds = useMemo(() => new Set(doctor.overrides.map((o) => o.id)), [doctor.overrides]);
   796→
   797→  const errors = useMemo(() => validateWeeklySchedule(draft), [draft]);
   798→  const errorCount = Object.keys(errors).length;
   799→
   800→  const mergedDoctor: Doctor = useMemo(
   801→    () => ({ ...doctor, weeklySchedule: draft, offDays: draftOffDays, overrides: draftOverrides }),
   802→    [doctor, draft, draftOffDays, draftOverrides]
   803→  );
   804→
   805→  const conflicts = useMemo(
   806→    () =>
   807→      findConflictingAppointments(mergedDoctor, allAppointments).filter((a) => a.status !== "completed"),
   808→    [mergedDoctor, allAppointments]
   809→  );
   810→
   811→  const previewSlots = useMemo(
   812→    () => (previewDate ? generateSlotsForDate(mergedDoctor, previewDate, allAppointments) : []),
   813→    [mergedDoctor, previewDate, allAppointments]
   814→  );
   815→
   816→  const previewOffDay = previewDate ? findOffDay(mergedDoctor, previewDate) : undefined;
   817→  const previewOverride = previewDate ? mergedDoctor.overrides.find((o) => o.date === previewDate) : undefined;
   818→
   819→  /* draft mutators */
   820→
   821→  const toggleDay = (day: number, active: boolean) => {
   822→    setDraft((prev) => {
   823→      const current = prev[day] ?? { active: false, ranges: [] };
   824→      return { ...prev, [day]: { ...current, active, ranges: active && current.ranges.length === 0 ? [newRange()] : current.ranges } };
   825→    });
   826→  };
   827→
   828→  const addRange = (day: number) => {
   829→    setDraft((prev) => {
   830→      const current = prev[day] ?? { active: true, ranges: [] };
   831→      if (current.ranges.length >= DRAFT_MAX_RANGES) {
   832→        toast.error(`حداکثر ${toFa(DRAFT_MAX_RANGES)} بازه در هر روز مجاز است`);
   833→        return prev;
   834→      }
   835→      return { ...prev, [day]: { ...current, ranges: [...current.ranges, newRange()] } };
   836→    });
   837→  };
   838→
   839→  const removeRange = (day: number, rangeId: string) => {
   840→    setDraft((prev) => {
   841→      const current = prev[day];
   842→      if (!current) return prev;
   843→      return { ...prev, [day]: { ...current, ranges: current.ranges.filter((r) => r.id !== rangeId) } };
   844→    });
   845→  };
